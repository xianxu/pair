// Scratch-only policy fault injector for #528. Build inside pair/cmd so Go's
// internal import rule applies. This program never stops a process or session.
package main

import (
	"bytes"
	"context"
	"encoding/json"
	"errors"
	"flag"
	"fmt"
	"os"
	"os/exec"
	"path/filepath"
	"reflect"
	"strconv"
	"strings"
	"syscall"
	"time"

	"github.com/xianxu/pair/cmd/internal/artifactpath"
	"github.com/xianxu/pair/cmd/internal/couchmessage"
	"github.com/xianxu/pair/cmd/internal/launcher"
	"github.com/xianxu/pair/cmd/internal/osfs"
	"github.com/xianxu/pair/cmd/internal/procutil"
	"github.com/xianxu/pair/cmd/internal/readiness"
	"github.com/xianxu/pair/cmd/internal/strictjson"
)

type saved struct {
	Version                          int
	Mode, Repo, DataRoot, PolicyPath string
	Actor                            couchmessage.Binding
	Policy                           artifactpath.SessionLaunchPolicy
	Original, Injected               []byte
}

func main() {
	if err := run(); err != nil {
		fmt.Fprintln(os.Stderr, "REFUSED:", err)
		os.Exit(1)
	}
}
func run() error {
	actorPath := flag.String("actor", "", "one exact .Actors[] JSON object")
	repoArg := flag.String("repo", "", "physical recovery-528-* scratch checkout")
	rootArg := flag.String("data-root", "", "normal GLOBAL Pair data root (not repos/scope)")
	mode := flag.String("mode", "", "legacy, conflict, or restore")
	backupArg := flag.String("backup", "", "new backup directory INSIDE scratch repo; restore reuses it")
	flag.Parse()
	if flag.NArg() != 0 || *actorPath == "" || *backupArg == "" {
		return errors.New("required: -actor -repo -data-root -mode -backup")
	}
	if *mode != "legacy" && *mode != "conflict" && *mode != "restore" {
		return errors.New("mode must be legacy, conflict, or restore")
	}
	ctx, cancel := context.WithTimeout(context.Background(), 20*time.Second)
	defer cancel()
	repo, err := physical(*repoArg)
	if err != nil {
		return err
	}
	if !strings.HasPrefix(filepath.Base(repo), "recovery-528-") || len(filepath.Base(repo)) <= len("recovery-528-") {
		return errors.New("repository must be a new recovery-528-* scratch checkout")
	}
	root, err := physical(*rootArg)
	if err != nil {
		return err
	}
	raw, err := osfs.ReadRegularMetadata(ctx, *actorPath, 64<<10)
	if err != nil {
		return err
	}
	var actor couchmessage.Candidate
	if err := strictjson.Decode(raw, &actor); err != nil {
		return fmt.Errorf("actor object: %w", err)
	}
	a := actor.Binding
	if err := a.Validate(); err != nil {
		return err
	}
	if a.Slot != filepath.Base(repo)+":0" {
		return errors.New("actor must be scratch repository's :0")
	}
	git := func(arg string) (string, error) {
		c := exec.CommandContext(ctx, "git", "-C", repo, "rev-parse", arg)
		b, e := c.Output()
		if e != nil {
			return "", e
		}
		p := strings.TrimSpace(string(b))
		if !filepath.IsAbs(p) {
			p = filepath.Join(repo, p)
		}
		return physical(p)
	}
	top, err := git("--show-toplevel")
	if err != nil || top != repo {
		return errors.New("scratch checkout is not its own Git root")
	}
	common, err := git("--git-common-dir")
	if err != nil {
		return err
	}
	actorRepo, err := physical(a.Repository)
	if err != nil || actorRepo != common {
		return errors.New("actor Git identity does not match scratch repository")
	}
	backup := filepath.Clean(*backupArg)
	if !filepath.IsAbs(backup) {
		return errors.New("backup must be absolute")
	}
	parent, err := physical(filepath.Dir(backup))
	if err != nil {
		return err
	}
	if parent != repo {
		return errors.New("backup must be a direct child of physical scratch repo")
	}
	backup = filepath.Join(parent, filepath.Base(backup))
	var state saved
	if *mode == "restore" {
		if p, e := physical(backup); e != nil || p != backup {
			return errors.New("backup directory missing or indirect")
		}
		b, e := osfs.ReadRegularMetadata(ctx, filepath.Join(backup, "manifest.json"), 64<<10)
		if e != nil {
			return e
		}
		if e = strictjson.Decode(b, &state); e != nil {
			return e
		}
		if state.Version != 1 || state.Repo != repo || state.DataRoot != root || !reflect.DeepEqual(state.Actor, a) || (state.Mode != "legacy" && state.Mode != "conflict") {
			return errors.New("backup owner/mode does not match exact request")
		}
		p, e := artifactpath.DecodeSessionLaunchPolicy(state.Original)
		if e != nil || p != state.Policy {
			return errors.New("backup policy mismatch")
		}
	} else {
		safety, e := launcher.ObserveTerminalSafety(ctx, a.Session, root, a.Scope, a.Tag)
		if e != nil {
			return e
		}
		if !safety.Usable() {
			return fmt.Errorf("source terminal not usable: %s", safety.Reason)
		}
		state = saved{Version: 1, Mode: *mode, Repo: repo, DataRoot: root, Actor: a, Policy: safety.Policy}
	}
	p := state.Policy
	if p.Scope != a.Scope || p.Tag != a.Tag || p.Name != a.Session || p.Agent != a.Agent || p.Nonce != a.Nonce {
		return errors.New("policy does not name exact actor")
	}
	socket, err := physical(p.Socket)
	if err != nil || socket != p.Socket {
		return errors.New("policy socket is not canonical")
	}
	cache, err := artifactpath.ResolvePairCache(os.Getenv("HOME"))
	if err != nil {
		return err
	}
	policyPath, err := cache.SessionPolicy(socket)
	if err != nil {
		return err
	}
	if *mode == "restore" && state.PolicyPath != policyPath {
		return errors.New("backup policy path mismatch")
	}
	state.PolicyPath = policyPath
	lockPath, err := cache.TerminalLease(socket)
	if err != nil {
		return err
	}
	lease, err := launcher.AcquireTerminalLease(ctx, lockPath)
	if err != nil {
		return err
	}
	defer lease.Close()
	check := func() error {
		if err := ctx.Err(); err != nil {
			return err
		}
		owner, e := (launcher.SessionOwnerProbe{}).Probe(ctx, a.Session, root, a.Scope, a.Tag)
		if e != nil {
			return e
		}
		if owner.State != launcher.SessionOwnerOwned || owner.Server.PID != p.ServerPID || owner.Server.Identity != p.ServerIdentity || owner.Server.Session != p.Name {
			return errors.New("server generation/ownership changed")
		}
		observed, e := physical(owner.Server.Socket)
		if e != nil || observed != socket {
			return errors.New("server socket changed")
		}
		info, e := os.Lstat(socket)
		if e != nil {
			return e
		}
		st, ok := info.Sys().(*syscall.Stat_t)
		if !ok || info.Mode()&os.ModeSocket == 0 || uint64(st.Dev) != p.SocketDevice || uint64(st.Ino) != p.SocketInode {
			return errors.New("socket identity changed")
		}
		if procutil.Identity(strconv.Itoa(a.PID)) != a.Start {
			return errors.New("wrapper process generation changed")
		}
		paths, e := artifactpath.Resolve(artifactpath.Address{DataDir: root, RepoScope: a.Scope, Tag: a.Tag})
		if e != nil {
			return e
		}
		b, e := osfs.ReadRegularMetadata(ctx, paths.AgentReady(a.Agent), 8192)
		if e != nil {
			return e
		}
		ready, e := readiness.Decode(string(b))
		if e != nil {
			return e
		}
		if ready.Agent != a.Agent || ready.Tag != a.Tag || ready.Session != a.Session || ready.Nonce != a.Nonce || ready.AgentIdentity == "" || ready.LaunchOrdinal == 0 {
			return errors.New("ready actor nonce/identity mismatch")
		}
		table, e := procutil.Table()
		if e != nil {
			return e
		}
		rows := map[int]procutil.Process{}
		for _, row := range table {
			rows[row.PID] = row
		}
		agent := rows[ready.PID]
		if agent.Identity != ready.AgentIdentity {
			return errors.New("native agent generation changed")
		}
		seen := map[int]bool{}
		for agent.PID > 0 && !seen[agent.PID] {
			if agent.PID == a.PID && agent.Identity == a.Start {
				return nil
			}
			seen[agent.PID] = true
			agent = rows[agent.PPID]
		}
		return errors.New("ready agent is not descendant of exact wrapper")
	}
	if err = check(); err != nil {
		return err
	}
	readPolicy := func() ([]byte, error) {
		return osfs.ReadRegularMetadata(ctx, policyPath, artifactpath.MaxSessionLaunchPolicyBytes)
	}
	if *mode != "restore" {
		safety, e := launcher.ObserveTerminalSafety(ctx, a.Session, root, a.Scope, a.Tag)
		if e != nil {
			return e
		}
		if !safety.Usable() || safety.Policy != p {
			return errors.New("usable policy changed while acquiring lease")
		}
		state.Original, err = readPolicy()
		if err != nil {
			return err
		}
		original, e := artifactpath.DecodeSessionLaunchPolicy(state.Original)
		if e != nil || original != p {
			return errors.New("policy bytes changed")
		}
		if *mode == "conflict" {
			state.Injected, err = conflictBytes(state.Original)
			if err != nil {
				return err
			}
		}
		if err = os.Mkdir(backup, 0700); err != nil {
			return fmt.Errorf("backup must be new/exclusive: %w", err)
		}
		manifest, e := json.MarshalIndent(state, "", "  ")
		if e != nil {
			return e
		}
		if err = exclusive(filepath.Join(backup, "manifest.json"), manifest); err != nil {
			return err
		}
		if err = exclusive(filepath.Join(backup, "policy.original.json"), state.Original); err != nil {
			return err
		}
		if err = syncDirectory(backup); err != nil {
			return err
		}
		if err = syncDirectory(filepath.Dir(backup)); err != nil {
			return err
		}
		if err = check(); err != nil {
			return err
		}
		current, e := readPolicy()
		if e != nil || !bytes.Equal(current, state.Original) {
			return errors.New("policy changed after backup; not modified")
		}
		if *mode == "legacy" {
			err = os.Remove(policyPath)
		} else {
			err = atomicWrite(policyPath, state.Injected, false)
		}
	} else {
		expected, e := conflictBytes(state.Original)
		if e != nil {
			return e
		}
		if state.Mode == "conflict" && !bytes.Equal(expected, state.Injected) {
			return errors.New("backup injected bytes invalid")
		}
		current, e := readPolicy()
		if state.Mode == "legacy" {
			if !errors.Is(e, os.ErrNotExist) {
				return errors.New("legacy restore requires policy absent; no overwrite")
			}
		} else if e != nil || !bytes.Equal(current, state.Injected) {
			return errors.New("conflict policy changed; no overwrite")
		}
		if err = check(); err != nil {
			return err
		}
		// Under the existing lease, replace only the exact injected value. Legacy
		// restoration uses link/no-replace even if a noncooperating writer appears.
		if state.Mode == "conflict" {
			current, e = readPolicy()
			if e != nil || !bytes.Equal(current, state.Injected) {
				return errors.New("policy changed before restore")
			}
		}
		err = atomicWrite(policyPath, state.Original, state.Mode == "legacy")
	}
	if err != nil {
		return err
	}
	if err = syncDirectory(filepath.Dir(policyPath)); err != nil {
		return fmt.Errorf("policy changed but directory sync failed: %w", err)
	}
	fmt.Printf("%s complete for %s; backup %s\n", *mode, a.Slot, backup)
	return nil
}
func physical(p string) (string, error) {
	if !filepath.IsAbs(p) {
		return "", errors.New("absolute path required")
	}
	return filepath.EvalSymlinks(filepath.Clean(p))
}
func conflictBytes(raw []byte) ([]byte, error) {
	var m map[string]json.RawMessage
	if err := json.Unmarshal(raw, &m); err != nil {
		return nil, err
	}
	var old string
	if err := json.Unmarshal(m["server_identity"], &old); err != nil {
		return nil, err
	}
	m["server_identity"], _ = json.Marshal(old + "-pair528-synthetic-conflict")
	return json.Marshal(m)
}
func exclusive(path string, b []byte) error {
	f, e := os.OpenFile(path, os.O_CREATE|os.O_EXCL|os.O_WRONLY, 0600)
	if e != nil {
		return e
	}
	_, e = f.Write(b)
	if e == nil {
		e = f.Sync()
	}
	return errors.Join(e, f.Close())
}
func atomicWrite(path string, b []byte, noReplace bool) error {
	f, e := os.CreateTemp(filepath.Dir(path), ".pair528-policy-")
	if e != nil {
		return e
	}
	name := f.Name()
	defer os.Remove(name)
	if _, e = f.Write(b); e == nil {
		e = f.Sync()
	}
	e = errors.Join(e, f.Close())
	if e != nil {
		return e
	}
	if noReplace {
		return os.Link(name, path)
	}
	return os.Rename(name, path)
}

func syncDirectory(path string) error {
	f, e := os.Open(path)
	if e != nil {
		return e
	}
	return errors.Join(f.Sync(), f.Close())
}
