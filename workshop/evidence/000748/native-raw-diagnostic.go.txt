package couchcore
import("context";"encoding/binary";"encoding/hex";"encoding/json";"fmt";"io";"net";"os";"path/filepath";"testing";"time";"github.com/xianxu/pair/cmd/internal/pairlifecycletest")
func TestRecoveryNativeRawDiagnostic748(t *testing.T){
 setupRecoveryLiveEnvironment(t)
 ctx,cancel:=context.WithTimeout(t.Context(),30*time.Second);defer cancel()
 session:=fmt.Sprintf("p748-%d",os.Getpid())
 f,err:=pairlifecycletest.StartControlledZellij(ctx,session);if err!=nil{t.Fatal(err)};defer f.Close()
 paths,err:=filepath.Glob(filepath.Join(os.Getenv("ZELLIJ_SOCKET_DIR"),"*",session));if err!=nil||len(paths)!=1{t.Fatalf("sockets=%v err=%v",paths,err)}
 log,err:=os.Create("EVIDENCE_JSONL");if err!=nil{t.Fatal(err)};defer log.Close()
 query,_:=hex.DecodeString("4a0b0a07920604280130012001")
 send:=func(c net.Conn,p []byte)error{frame:=make([]byte,4+len(p));binary.LittleEndian.PutUint32(frame,uint32(len(p)));copy(frame[4:],p);_,err:=c.Write(frame);return err}
 bad,crossed:=0,0
 for sample:=0;sample<2000;sample++{
  probe:=sample%2==0;payload:=query;if probe{payload=[]byte{0x6a,0}}
  c,err:=net.DialTimeout("unix",paths[0],time.Second);var reply []byte
  if err==nil{_=c.SetDeadline(time.Now().Add(time.Second));err=send(c,payload)
   if err==nil{var h [4]byte;_,err=io.ReadFull(c,h[:]);if err==nil{n:=binary.LittleEndian.Uint32(h[:]);if n>1048576{err=fmt.Errorf("oversize %d",n)}else{reply=make([]byte,n);_,err=io.ReadFull(c,reply)}}}
   if !probe{_=send(c,[]byte{0x5a,0})};_=c.Close()
  }
  ok:=err==nil&&len(reply)>0&&((probe&&reply[0]==0x22)||(!probe&&reply[0]==0x2a))
  if !ok{bad++;if len(reply)>0&&((probe&&reply[0]==0x2a)||(!probe&&reply[0]==0x22)){crossed++}}
  shown:=reply;if len(shown)>200{shown=shown[:200]};_=json.NewEncoder(log).Encode(map[string]any{"sample":sample,"probe":probe,"ok":ok,"error":fmt.Sprint(err),"reply":hex.EncodeToString(shown)})
  if ctx.Err()!=nil{t.Fatal(ctx.Err())}
 }
 t.Logf("2000 independent sequential requests: bad-first-reply=%d crossed=%d",bad,crossed)
 if bad>0{t.Fatal("native first-response corruption reproduced; no retries")}
}
